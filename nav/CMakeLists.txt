cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(nav_core VERSION 0.1.0 LANGUAGES C CXX)
  set(CMAKE_C_STANDARD 99)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
endif()
add_library(navcore src/strapdown.cpp src/eskf.cpp src/nav_core.cpp)
target_include_directories(navcore PUBLIC include)
add_library(navprotocol src/protocol.c)
add_library(nav_protocol ALIAS navprotocol)
target_include_directories(navprotocol PUBLIC include)
if(MSVC)
  target_compile_options(navcore PRIVATE /W4 /GR- /utf-8)
  target_compile_options(navprotocol PRIVATE /W4 /utf-8)
else()
  target_compile_options(navcore PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
  target_compile_options(navprotocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(navprotocol PUBLIC m)
endif()
if(NOT CMAKE_CROSSCOMPILING)
  add_executable(nav_sim sim/sim_main.cpp)
  target_link_libraries(nav_sim PRIVATE navcore navprotocol)
  add_executable(nav_replay apps/replay_main.cpp)
  target_link_libraries(nav_replay PRIVATE navcore)
  add_executable(nav_mcu_host apps/mcu_main.cpp)
  target_link_libraries(nav_mcu_host PRIVATE navcore navprotocol)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE navcore)
  add_test(NAME nav_core COMMAND test_core)
  add_executable(test_protocol tests/test_protocol.c)
  target_link_libraries(test_protocol PRIVATE navprotocol)
  add_test(NAME nav_protocol COMMAND test_protocol)
  add_executable(test_aux tests/test_aux.cpp)
  target_link_libraries(test_aux PRIVATE navcore)
  add_test(NAME nav_aux COMMAND test_aux)
  add_executable(test_integrity tests/test_integrity.cpp)
  target_link_libraries(test_integrity PRIVATE navcore)
  add_test(NAME nav_integrity COMMAND test_integrity)
  add_test(NAME nav_sim_smoke COMMAND nav_sim 60 1 ${CMAKE_CURRENT_BINARY_DIR}/smoke combined)
  add_test(NAME nav_mcu_host_smoke COMMAND nav_mcu_host)
  add_test(NAME nav_replay_smoke COMMAND nav_replay ${CMAKE_CURRENT_BINARY_DIR}/smoke/events.csv ${CMAKE_CURRENT_BINARY_DIR}/smoke/replayed.csv)
  set_tests_properties(nav_replay_smoke PROPERTIES DEPENDS nav_sim_smoke)
endif()
