cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(uav_companion LANGUAGES CXX)
    include(CTest)
endif()

add_library(companion_lib src/companion.cpp src/matching.cpp src/terrain.cpp src/tile_store.cpp)
target_include_directories(companion_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(companion_lib PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(companion_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(nav_companion src/main.cpp)
target_link_libraries(nav_companion PRIVATE companion_lib)

if(BUILD_TESTING)
    add_executable(test_companion tests/test_companion.cpp)
    target_link_libraries(test_companion PRIVATE companion_lib)
    add_test(NAME companion_algorithms COMMAND test_companion)
    add_test(NAME companion_demo COMMAND nav_companion --demo)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME reference_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../tools/test_reference.py --companion $<TARGET_FILE:nav_companion>)
    endif()
endif()
