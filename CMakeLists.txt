cmake_minimum_required(VERSION 3.20)
project(uav_navigation2 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(NAV_BUILD_COMPANION "Build portable CPU visual navigation service" ON)
option(NAV_BUILD_FIRMWARE "Build portable MCU runtime and sensor drivers" ON)
option(NAV_SANITIZE "Enable address/undefined sanitizers for host verification" OFF)
if(NAV_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(nav)
if(NAV_BUILD_COMPANION)
  add_subdirectory(companion)
endif()
if(NAV_BUILD_FIRMWARE)
  add_subdirectory(firmware)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/integration/CMakeLists.txt")
  add_subdirectory(integration)
endif()
if(NAV_BUILD_COMPANION AND NAV_BUILD_FIRMWARE AND NOT CMAKE_CROSSCOMPILING)
  add_executable(nav_pipeline_demo apps/pipeline_demo.cpp)
  target_link_libraries(nav_pipeline_demo PRIVATE navfirmware companion_lib)
  if(BUILD_TESTING)
    add_test(NAME navigation_pipeline COMMAND nav_pipeline_demo)
  endif()
endif()
find_package(Python3 QUIET COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME replay_equivalence COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_replay.py ${CMAKE_CURRENT_BINARY_DIR}/nav/smoke/estimate.csv ${CMAKE_CURRENT_BINARY_DIR}/nav/smoke/replayed.csv)
  set_tests_properties(replay_equivalence PROPERTIES DEPENDS nav_replay_smoke)
endif()
